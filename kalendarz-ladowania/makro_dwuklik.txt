' ============================================================
'  MAKRO "DWUKLIK = NAŁADOWANE" – Kalendarz ładowania akumulatorów
' ============================================================
'  Jak dodać (jednorazowo, ok. 1 minuty):
'   1. Otwórz plik w Excelu, naciśnij Alt + F11 (otworzy się edytor VBA).
'   2. Po lewej kliknij dwa razy arkusz "Kalendarz" (np. "Arkusz2 (Kalendarz)").
'   3. Wklej poniższy kod (od "Private Sub" do "End Sub") i zamknij edytor.
'   4. Plik > Zapisz jako > "Skoroszyt programu Excel z obsługą makr (*.xlsm)".
'   5. Przy otwieraniu pliku kliknij "Włącz zawartość".
'  Od teraz: dwuklik w kratce = ✔ (naładowane). Ponowny dwuklik = usunięcie.
' ============================================================

Private Sub Worksheet_BeforeDoubleClick(ByVal Target As Range, Cancel As Boolean)
    Dim siatka As Range
    Set siatka = ThisWorkbook.Names("KalendarzSiatka").RefersToRange
    If Intersect(Target, siatka) Is Nothing Then Exit Sub
    If Target.Cells.Count > 1 Then Exit Sub
    Cancel = True
    If Me.Cells(Target.Row, 2).Value = "" Then Exit Sub   ' brak auta w tym wierszu
    If Target.Value = "" Then
        Target.Value = ChrW(10004)   ' ✔
    Else
        Target.ClearContents
    End If
End Sub
